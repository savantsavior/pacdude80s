---------------------------------------------------------
function ApplyScreenFade()
	if ScrFadeStatus==FadeIdle then
		return
	end

 if ScrFadeStatus==FadeIn then
 	if ScrFadeScale>0 then
  	ScrFadeScale=ScrFadeScale-1
		else
			ScrFadeStatus=FadeIdle
		end
 elseif ScrFadeStatus==FadeOut then
 	if ScrFadeScale<32 then
  	ScrFadeScale=ScrFadeScale+1
		else
			ScrFadeStatus=FadeIn
			Scr=ScrNext

			if Scr==3 and ClearedBoard==0 then
				ResetGameAfterClear()
				DelayInput=25
			end

			if Scr==3 and PacDudeKilled==0 then
				PacDudeKilled=-1
				ResetGameAfterDeath()
				DelayInput=25
			end
		end
	end

	spr(0,120-(4*ScrFadeScale),68-(4*ScrFadeScale),-1,ScrFadeScale,0,0,1,1)
end

function DrawText(text,x,y,just,fillclr,outlineclr,fixed,scale,smallFont)
	local xOffset=0
	local width=print(text,0,-999,outlineclr,fixed,scale,smallFont)
	if just==0 then xOffset=x end
	if just==1 then xOffset=(240/2)-(width/2) end
	if just==2 then xOffset=(240-width)-x end

	local xOff
	local yOff
	for yOff=y-1,y+1 do
		for xOff=xOffset-1,xOffset+1 do
			if (xOff~=0 and yOff~=0) then
				print(text,xOff,yOff,outlineclr,fixed,scale,smallFont)
			end
		end
	end

	print(text,xOffset,y,fillclr,fixed,scale,smallFont)
end

function CheckAllColl(tileX,tileY,x,xMax,y,yMax)
	local stepY=1
	if y>yMax then stepY=-1 end
	local stepX=1
	if x>xMax then stepX=-1 end

	for testY=y,yMax,stepY do
		for testX= x,xMax,stepX do
			TileIndex=mget( (tileX+testX),(tileY+testY) )
			if fget(TileIndex,0)==true then
			 return true
			end
		end
	end

	return false
end

---------------------------------------------------------
function Collision(tileX,tileY,dir)
	if dir==MUp then
	 if CheckAllColl(tileX,tileY,0,1,0,-1)==true then
			return true
		end
	elseif dir==MDown then
	 if CheckAllColl(tileX,tileY,0,1,0,2)==true then
			return true
		end
	elseif dir==MLeft then
	 if CheckAllColl(tileX,tileY,0,-1,0,1)==true then
			return true
		end
	elseif dir==MRight then
	 if CheckAllColl(tileX,tileY,0,2,0,1)==true then
			return true
		end
	end

	return false
end

function AnimatePowerPellets()

		PowerFlash=PowerFlash+1
	else
		PowerFlash=0
	end

	for x=1,58+1 do
	 for y=1,34+1 do
			if PowerFlash<14 then
	 	 if Board[x][y]==36 then mset(x-1,y-1,34) end
	 	 if Board[x][y]==37 then mset(x-1,y-1,35) end
	 	 if Board[x][y]==52 then mset(x-1,y-1,50) end
	 	 if Board[x][y]==53 then mset(x-1,y-1,51) end
			elseif PowerFlash>13 then
	 	 if Board[x][y]==34 then mset(x-1,y-1,36) end
	 	 if Board[x][y]==35 then mset(x-1,y-1,37) end
	 	 if Board[x][y]==50 then mset(x-1,y-1,52) end
	 	 if Board[x][y]==51 then mset(x-1,y-1,53) end
			end
		end
	end
end

---------------------------------------------------------
function CheckPlayerCollisions()
	CollUp=false
	CollDown=false
	CollLeft=false
	CollRight=false

	if Collision(MapTileX,MapTileY,MUp)==true then
		CollUp=true
	end
	if Collision(MapTileX,MapTileY,MDown)==true then
		CollDown=true
	end
	if Collision(MapTileX,MapTileY,MLeft)==true then
		CollLeft=true
	end
	if Collision(MapTileX,MapTileY,MRight)==true then
		CollRight=true
	end

	for ghost=1,6 do
		if PlayerScreenY>GhostScreenY[ghost]-12 and PlayerScreenY<GhostScreenY[ghost]+12 then
			if PlayerScreenX>GhostScreenX[ghost]-12 and PlayerScreenX<GhostScreenX[ghost]+12 then
				if GhostStatus[ghost]==Attack then
					PacDudeKilled=40
				elseif GhostStatus[ghost]==Scared then
					GhostNumberEaten=GhostNumberEaten+1
					Score=Score+100+(250*GhostNumberEaten)
					GhostStatus[ghost]=Eyes
					sfx(2,43,15,2,15-7,0)
				end
			end
		end
	end
end

function MovePlayer()

function MoveGhosts()
	if ClearedBoard>-1 then return end

	for ghost=1,6 do
		if GhostMoveX[ghost]==0 and GhostMoveY[ghost]==0 and DelayInput==0 then
			GhostCollUp[ghost]=false
			GhostCollRight[ghost]=false
			GhostCollDown[ghost]=false
			GhostCollLeft[ghost]=false

			if Collision(GhostTileX[ghost],GhostTileY[ghost],MUp)==true then
				GhostCollUp[ghost]=true
			end
			if Collision(GhostTileX[ghost],GhostTileY[ghost],MDown)==true then
				GhostCollDown[ghost]=true
			end
			if Collision(GhostTileX[ghost],GhostTileY[ghost],MLeft)==true then
			 GhostCollLeft[ghost]=true
			end
			if Collision(GhostTileX[ghost],GhostTileY[ghost],MRight)==true then
				GhostCollRight[ghost]=true
			end

			local foundMove=false
			while foundMove==false do
				local randomDir=math.random(0,4)

				if GhostTileX[ghost]==28 and GhostTileY[ghost]==16 then
					if GhostStatus[ghost]==Eyes then
						GhostStatus[ghost]=Attack
						GhostScaredTimer[ghost]=0
					end

					GhostMoveY[ghost]=-1
					GhostDir[ghost]=MUp
					foundMove=true
				end

				if foundMove==false then
					if GhostStatus[ghost]==Attack then
						if GhostScreenX[ghost]==PlayerScreenX then
							if GhostScreenY[ghost]>PlayerScreenY and GhostCollUp[ghost]==false then
								GhostMoveY[ghost]=-1
								GhostDir[ghost]=MUp
								foundMove=true
							elseif GhostScreenY[ghost]<PlayerScreenY and GhostCollDown[ghost]==false then
								GhostMoveY[ghost]=1
								GhostDir[ghost]=MDown
								foundMove=true
							end
						elseif GhostScreenY[ghost]==PlayerScreenY then
							if GhostScreenX[ghost]>PlayerScreenX and GhostCollLeft[ghost]==false then
								GhostMoveX[ghost]=-1
								GhostDir[ghost]=MLeft
								foundMove=true
							elseif GhostScreenX[ghost]<PlayerScreenX and GhostCollRight[ghost]==false then
								GhostMoveX[ghost]=1
								GhostDir[ghost]=MRight
								foundMove=true
							end
						end
					elseif GhostStatus[ghost]==Eyes then
						local gY=GhostTileY[ghost]
						local gX=GhostTileX[ghost]

						if BoardReturn[gX][gY]==240 then
							GhostMoveY[ghost]=-1
							GhostDir[ghost]=MUp
							foundMove=true
						elseif BoardReturn[gX][gY]==242 then
							GhostMoveY[ghost]=1
							GhostDir[ghost]=MDown
							foundMove=true
						elseif BoardReturn[gX][gY]==243 then
							GhostMoveX[ghost]=-1
							GhostDir[ghost]=MLeft
							foundMove=true
						elseif BoardReturn[gX][gY]==241 then
							GhostMoveX[ghost]=1
							GhostDir[ghost]=MRight
							foundMove=true
						elseif BoardReturn[gX][gY]==245 then
							local randomHorizontal=math.random(1, 2)
							if randomHorizontal==1 then
								GhostMoveX[ghost]=-1
								GhostDir[ghost]=MLeft
								foundMove=true
							elseif randomHorizontal==2 then
								GhostMoveX[ghost]=1
								GhostDir[ghost]=MRight
								foundMove=true
							end
						end
					end
				end

				if foundMove==false then
					if randomDir==0 and GhostCollUp[ghost]==false and GhostDir[ghost]~=MDown then
						GhostMoveY[ghost]=-1
						GhostDir[ghost]=MUp
						foundMove=true
					end
					if randomDir==1 and GhostCollRight[ghost]==false and GhostDir[ghost]~=MLeft then
						GhostMoveX[ghost]=1
						GhostDir[ghost]=MRight
						foundMove=true
					end
					if randomDir==2 and GhostCollDown[ghost]==false and GhostDir[ghost]~=MUp then
						GhostMoveY[ghost]=1
						GhostDir[ghost]=MDown
						foundMove=true
					end
					if randomDir==3 and GhostCollLeft[ghost]==false and GhostDir[ghost]~=MRight then
						GhostMoveX[ghost]=-1
						GhostDir[ghost]=MLeft
						foundMove=true
					end
				end
			end
		end

		if (GhostMoveY[ghost]>-16 and GhostMoveY[ghost]<0) then
			GhostMoveY[ghost]=GhostMoveY[ghost]-1
		elseif GhostMoveY[ghost]==-16 then
			GhostMoveY[ghost]=0
			GhostScreenY[ghost]=GhostScreenY[ghost]-16
			GhostTileY[ghost]=GhostTileY[ghost]-2
		end

		if (GhostMoveY[ghost]<16 and GhostMoveY[ghost]>0) then
			GhostMoveY[ghost]=GhostMoveY[ghost]+1
		elseif GhostMoveY[ghost]==16 then
			GhostMoveY[ghost]=0
			GhostScreenY[ghost]=GhostScreenY[ghost]+16
			GhostTileY[ghost]=GhostTileY[ghost]+2
		end

		if (GhostMoveX[ghost]>-16 and GhostMoveX[ghost]<0) then
			GhostMoveX[ghost]=GhostMoveX[ghost]-1
		elseif GhostMoveX[ghost]==-16 then
			GhostMoveX[ghost]=0
			GhostScreenX[ghost]=GhostScreenX[ghost]-16
			GhostTileX[ghost]=GhostTileX[ghost]-2
		end

		if (GhostMoveX[ghost]<16 and GhostMoveX[ghost]>0) then
			GhostMoveX[ghost]=GhostMoveX[ghost]+1
		elseif GhostMoveX[ghost]==16 then
			GhostMoveX[ghost]=0
			GhostScreenX[ghost]=GhostScreenX[ghost]+16
			GhostTileX[ghost]=GhostTileX[ghost]+2
		end
	end
end

function DrawGhosts()
	if Pause==false then
		if GhostDrawOrder<1 then
			GhostDrawOrder=GhostDrawOrder+1
		else
			GhostDrawOrder=0
		end

		if GhostAnimeTimer<6 then
			GhostAnimeTimer=GhostAnimeTimer+1
		else
			GhostAnimeTimer=0
		end

		local aGhostIsScared=false
		for ghost=1,6 do
			if GhostScaredTimer[ghost]>0 then
				aGhostIsScared=true
				GhostScaredTimer[ghost]=GhostScaredTimer[ghost]-1
			end

			if	GhostScaredTimer[ghost]==1 then
				GhostScaredTimer[ghost]=0

				GhostNumberEaten=0

				if GhostStatus[ghost]==Scared then
					GhostStatus[ghost]=Attack
				end
			end
		end

		if aGhostIsScared==true then
			if GhostScaredEffectTimer>0 then
				if GhostScaredEffectTimer%25==0 then
					sfx(1,43,30,1,15-7,0)
				end

				GhostScaredEffectTimer=GhostScaredEffectTimer-1

				if GhostScaredEffectTimer==0 then GhostScaredEffectTimer=100 end
			end
		else
			sfx(-1,43,60,1,15-7,0)
		end

		for ghost=1,6 do
			if GhostEyeBlinkTimer[ghost]<100 then
				GhostEyeBlinkTimer[ghost]=GhostEyeBlinkTimer[ghost]+1
			else
				GhostEyeBlinkTimer[ghost]=0
			end
		end
	end

	local start=1
	local theend=6
	local step=1

	if GhostDrawOrder==1 then
		start=6
		theend=1
		step=-1
	end

	for ghost=start,theend,step do
		local ghostFrame=288+((ghost-1)*2)

		if GhostScaredTimer[ghost]>0 then ghostFrame=300 end

		if GhostAnimeTimer>2 then
			ghostFrame=320+((ghost-1)*2)

			if GhostScaredTimer[ghost]>50 then ghostFrame=332 end

			if (GhostScaredTimer[ghost]>0 and GhostScaredTimer[ghost]<51) and GhostScaredTimer[ghost]%2==0 then
				ghostFrame=332
			end
		end

		if GhostStatus[ghost] ~= Eyes then
			spr(ghostFrame,GhostScreenX[ghost]+GhostMoveX[ghost],GhostScreenY[ghost]+GhostMoveY[ghost],0,1,0,0,2,2)
		end

		local eyesDir=352
		if GhostStatus[ghost]==Attack or GhostStatus[ghost]==Eyes then
			if GhostDir[ghost]==MUp then eyesDir=352 end
			if GhostDir[ghost]==MRight then eyesDir=354 end
			if GhostDir[ghost]==MDown then eyesDir=356 end
			if GhostDir[ghost]==MLeft then eyesDir=358 end
		elseif GhostStatus[ghost]==Scared then
			if GhostDir[ghost]==MDown then eyesDir=352 end
			if GhostDir[ghost]==MLeft then eyesDir=354 end
			if GhostDir[ghost]==MUp then eyesDir=356 end
			if GhostDir[ghost]==MRight then eyesDir=358 end
		end

		if GhostEyeBlinkTimer[ghost]<90 then
			spr(eyesDir,GhostScreenX[ghost]+GhostMoveX[ghost],GhostScreenY[ghost]+GhostMoveY[ghost],0,1,0,0,2,2)
		end
	end
end

function ScreenPlaying()
	if	ClearedBoard>0 then
		ChangePaletteColor(13,95,176,194)

		ClearedBoard=ClearedBoard-1

		if ClearedBoard%10==0 or ClearedBoard%11==0 or ClearedBoard%12==0 or ClearedBoard%13==0 or ClearedBoard%14==0 then
			ChangePaletteColor(13,255,255,255)
		end

		if ClearedBoard==0 then
			ScrFadeStatus=FadeOut
			ScrNext=3
			ScrTime=400
			DelayInput=100
			Pause=false
			PauseOffset=0
		end
	end

	cls(0)

	if Pause==false and DelayInput==0 then
		CheckForDebug()

		CheckForBoardClear()

		if PacDudeKilled==-1 then
			CheckPlayerCollisions()
			MovePlayer()
			MoveGhosts()
		elseif PacDudeKilled>0 then
			PacDudeKilled=PacDudeKilled-1
			if PacDudeKilled==1 then
				ScrFadeStatus=FadeOut

				if Lives>0 then
					ScrNext=3
					Lives=Lives-1
				elseif Lives==0 then
					ScrNext=2

					if Score>HighScore then
						HighScore=Score
					end
					pmem(0,HighScore)
				end
			end
		end

		ReadBoard()

		AnimatePowerPellets()
	end

	map(0,0,30*2,18*2,MapScreenX,MapScreenY)

	DrawPlayer()

	if PacDudeKilled==-1 or PacDudeKilled>250 then
		DrawGhosts()
	end

	if Score>HighScore then
		if NewHighScoreAcheived==false then
			NewHighScoreAcheived=true
			sfx(4,43,200,3,15-7,0)
		end

		HighScore=Score
	end

	if Score>50000 and OneUp==false then
		OneUp=true
		Lives=Lives+1
		sfx(4,43,50,3,15-7,0)
	end

	DrawText("High Score: "..HighScore,0,2,1,4,0,false,1,true)
 DrawText("Score:"..Score,0,2+8,1,4,0,false,1,true)

 DrawText("X"..Lives,(240/2)+4,136-9,0,4,0,false,1,true)

	DisplayDebug()

	if key(48) and DelayInput==0 and PacDudeKilled==-1 and ClearedBoard==-1 then
		sfx(0,95-46,5,0,15-7,0)
		if Pause==false then
			Pause=true
			DelayInput=20

			if Debug~=1 then
				for index=0,15 do
					for indexTwo=0,50 do
						local base=0x3FC0+(index*3)
	    	local r=peek(base+0)
	    	local g=peek(base+1)
	    	local b=peek(base+2)

	     if r>0 then r=r-1 end
	     if g>0 then g=g-1 end
	     if b>0 then b=b-1 end
	     poke(base,r,g,b)
					end
				end
			end
		elseif Pause==true then

			DelayInput=20
			ResetPaletteColors()
		end
	end

	if Pause==true and Debug~=1 then
		DrawText("PAUSED!",0,(136/2)-10,1,4,15,false,2,true)
		DrawText("Press [Spacebar] On Keyboard To Play!",0,(136/2)+10,1,4,15,false,1,true)
	end
end

function TIC()
	FrameFPS=FrameFPS+1

	GetAllInput()

	if Scr==0 then
		ScreenMIT()
	elseif Scr==1 then
		ScreenTIC()
	elseif Scr==2 then
		ScreenTitle()
	elseif Scr==3 then
		ScreenPlaying()
	end

	ApplyScreenFade()

	TimeCurr = time()
	if TimeCurr>TimeEnd then
		FPS=FrameFPS
		TimeEnd = time()+1000
		FrameFPS=0
	end

	if Debug>0 and Pause==false then
	 DrawText("P"..PelletCount.." X"..MouseX.." Y"..MouseY.." F"..FPS,2,136-8,2,13,0,false,1,true)
	end

	DrawScreenOutline(12)
end
